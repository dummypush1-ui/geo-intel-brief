class Cursor:
 def __init__(self,client):self.c=client
 def sort(self,*v):self.c.calls.append(('sort',v));return self
 def limit(self,*v):self.c.calls.append(('limit',v));return self
 def max_time_ms(self,*v):self.c.calls.append(('max_time_ms',v));return self
 def __iter__(self):
  if self.c.fail:raise ValueError('secret-bearing simulated failure')
  return iter(copy.deepcopy(self.c.rows))
 def close(self):self.c.calls.append(('cursor_close',))
class Client:
 def __init__(self,rows=None):self.paths=[];self.calls=[];self.fail=False;self.closed=False;self.rows=rows if rows is not None else [dict(_id='PRIVATEID',id='OTHERID',url='https://example.com/story',title='Public article',summary='Public summary',source='Source',country='India',category='TRADE',created_at=datetime.now(timezone.utc),risk_level='HIGH',credibility='HIGH',score=4,emailed=True,telegram_url='https://private.example',password='secret')]
 def __getitem__(self,k):self.paths.append(k);return self
 def find(self,q,p):self.calls.append(('find',q,p));return Cursor(self)
 def close(self):self.closed=True
