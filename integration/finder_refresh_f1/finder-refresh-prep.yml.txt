  schedule:
    - cron: '30 3 * * *'
  workflow_dispatch:
permissions:
  contents: read
concurrency:
  group: merged-finder-refresh-prep
  cancel-in-progress: false
jobs:
  prepare:
    if: github.repository == 'dummypush1-ui/geo-intel-brief' && vars.MERGED_FINDER_REFRESH_PREP_ENABLED == 'true'
    runs-on: ubuntu-24.04
    timeout-minutes: 5
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262
        with:
          persist-credentials: false
      - name: Validate Node without installing or running the old updater
        run: node -e 'if(Number(process.versions.node.split(".")[0])<22)process.exit(1)'
      - name: Rebuild in temporary scratch and report consumer holds
        env:
          MERGED_FINDER_REFRESH_PREP_ENABLED: 'true'
        run: python3 -B -m integration.finder_refresh_f1
