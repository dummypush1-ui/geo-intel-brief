"""Fixed separate authored-CA local proof. Never production import/runtime option."""
import ssl,socket,threading,time,unittest
from http_closed import parse
class TLSProof(unittest.TestCase):
 def once(self,name='nilgiried.com'):
  ctx=ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER);ctx.load_cert_chain('fixture-server.pem','fixture-server-key.pem');sni=[];ctx.set_servername_callback(lambda sock,name,context:sni.append(name));server=socket.socket();server.bind(('127.0.0.1',0));server.listen(1);port=server.getsockname()[1];seen=[];errors=[]
  def run():
   conn=None
   try:
    conn,peer=server.accept();conn.settimeout(1)
    with ctx.wrap_socket(conn,server_side=True)as tls:
     request=bytearray()
     while not request.endswith(b'\r\n\r\n'):
      b=tls.recv(1)
      if not b:break
      request+=b
      if len(request)>4096:raise ValueError()
     seen.append(bytes(request));tls.sendall(b'HTTP/1.1 200 OK\r\nContent-Length: 5\r\nConnection: close\r\n\r\nHello')
   except (ssl.SSLError,OSError)as e:errors.append(type(e).__name__)
   finally:
    if conn is not None:conn.close()
    server.close()
  thread=threading.Thread(target=run);thread.start();clientctx=ssl.create_default_context(cafile='fixture-ca.pem');raw=socket.socket();raw.settimeout(1)
  try:
   raw.connect(('127.0.0.1',port));self.assertEqual(raw.getpeername()[0],'127.0.0.1')
   with clientctx.wrap_socket(raw,server_hostname=name)as tls:
    self.assertEqual(clientctx.verify_mode,ssl.CERT_REQUIRED);self.assertTrue(clientctx.check_hostname);self.assertIn('nilgiried.com',[n for typ,n in tls.getpeercert()['subjectAltName']]);tls.sendall(b'GET / HTTP/1.1\r\nHost: nilgiried.com\r\nConnection: close\r\n\r\n');result=parse(tls,time.monotonic()+1)
  finally:raw.close();thread.join(3);self.assertFalse(thread.is_alive())
  return result,sni,seen
 def test_actual_tls_peer_sni_host(self):
  r,s,h=self.once();self.assertEqual(r['body'],b'Hello');self.assertEqual(s,['nilgiried.com']);self.assertIn(b'Host: nilgiried.com\r\n',h[0])
 def test_wrong_certificate_name(self):
  with self.assertRaises(ssl.SSLCertVerificationError):self.once('wrong.invalid')
if __name__=='__main__':unittest.main(verbosity=2)
